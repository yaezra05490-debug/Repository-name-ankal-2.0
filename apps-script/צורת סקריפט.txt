 *
 * הפריסה מוגדרת "כל אחד", כדי שדף מקומי יוכל לפנות אליה. לכן ההגנה היא אסימון סודי:
 * כל בקשה חייבת לשאת את ה-SECRET, אחרת היא נדחית. שנה אותו לערך פרטי משלך,
 * וּודא שאותו ערך בדיוק מופיע גם בקובץ ה-HTML המקומי.

// >>> שנה את הערך הזה, ותכניס בדיוק אותו גם בקובץ ה-HTML המקומי <<<
var SECRET = "sbmgr-4Qw9Zx2Lp7Rt6Kv1Nc8Hj3Ff5Bd0MsYu7Aa";

var SCRIPT_MIME = "application/vnd.google-apps.script";

// ==========================================
// עוזרים: גישה לשירותים ובניית כתובות מחלקים
// ==========================================
function getAppService(partsArray) {
  var serviceName = partsArray.join("");
  var context = typeof globalThis !== "undefined" ? globalThis : this;
  return context[serviceName];
}

function buildEndpoint(parts) {
  return parts.join("");
}

function scriptApiBase() {
  return buildEndpoint(["https://", "script.google", "apis", ".com/", "v1/"]);
}

// ==========================================
// נקודת כניסה
// הדף עצמו מקומי (במחשב) ופונה לכאן ב-POST. פנייה ב-GS (פתיחת הכתובת בדפדפן)
// לא חושפת כלום - רק הודעת סטטוס.
// ==========================================
function doGet() {
  var out = getAppService(["C","o","n","t","e","n","t","S","e","r","v","i","c","e"]);
  return out.createTextOutput("מנהל הסקריפטים פעיל.").setMimeType(out.MimeType.TEXT);
}

function doPost(e) {
  var out = getAppService(["C","o","n","t","e","n","t","S","e","r","v","i","c","e"]);
  var result;
  try {
    var data = JSON.parse(e.postData.contents);
    if (!data || data.token !== SECRET) {
      result = { ok: false, error: "אסימון שגוי או חסר" };
    } else {
      result = route_(data);
    }
  } catch (err) {
    result = { ok: false, error: "שגיאת שרת: " + err.message };
  }
  return out.createTextOutput(JSON.stringify(result)).setMimeType(out.MimeType.JSON);
}

function route_(data) {
  switch (data.action) {
    case 'list':      return listScripts();
    case 'get':       return getProject(data.scriptId);
    case 'save':      return saveProject(data.scriptId, data.files);
    case 'deploy':    return deployProject(data.scriptId, data.deploymentId, data.description);
    case 'settings':  return getSettings();
    case 'setNotify': return setNotify(data.on);
    default:          return { ok: false, error: "פעולה לא מוכרת" };
  }
}

// ==========================================
// קריאה כללית ל-API עם ה-token של החשבון
// ==========================================
function api_(method, url, payload) {
  var scriptApp = getAppService(["S","c","r","i","p","t","A","p","p"]);
  var fetchApp  = getAppService(["U","r","l","F","e","t","c","h","A","p","p"]);

  var params = {
    method: method,
    headers: { Authorization: 'Bearer ' + scriptApp.getOAuthToken() },
    muteHttpExceptions: true
  };
  if (payload) {
    params.contentType = 'application/json';
    params.payload = JSON.stringify(payload);
  }
  var res = fetchApp.fetch(url, params);
  var text = res.getContentText();
  var body = null;
  try { body = text ? JSON.parse(text) : null; } catch (e) { body = { raw: text }; }
  return { code: res.getResponseCode(), body: body };
}

// זיהוי בעיות הרשאה / API כבוי, והחזרת הודעה ברורה עם הפתרון המדויק
function authProblem_(r) {
  if (r.code === 401) {
    return "צריך להתחבר מחדש ולאשר את ההרשאות של המנהל.";
  }
  if (r.code === 403) {
    var msg = (r.body && r.body.error && r.body.error.message) || "";
    // רק כשההודעה מדברת ספציפית על ה-Apps Script API - מציגים את הודעת המתג.
    // שגיאות 403 אחרות מוצגות כמות שהן, כדי לא לאבחן שגוי.
    if (/script\.googleapis|Apps Script API/i.test(msg)) {
      return "צריך להפעיל את \"Google Apps Script API\" בהגדרות החשבון: היכנס ל-script.google.com/home/usersettings והדלק את המתג, ואז נסה שוב.";
    }
    return "אין הרשאה (403): " + (msg || "אין פירוט מהשרת");
  }
  return null;
}

function fail_(subject, r) {
  var msg = (r && r.body && r.body.error && r.body.error.message) || ("קוד " + (r ? r.code : "?"));
  notify_(subject + " נכשל", msg);
  return { ok: false, error: msg };
}

// ==========================================
// רשימת כל הסקריפטים בחשבון
// דרך DriveApp המובנה - לא דרך ה-Drive API. גוגל-סקריפט רגיל אינו יכול לקרוא ל-Drive API
// (הוא מחזיר 403 "השירות לא הופעל"), אבל DriveApp הוא שירות פנימי שלא דורש הפעלת שום שירות.
// ==========================================
function listScripts() {
  try {
    var driveApp = getAppService(["D","r","i","v","e","A","p","p"]);
    var it = driveApp.searchFiles('mimeType = "' + SCRIPT_MIME + '" and trashed = false');
    var scripts = [];
    while (it.hasNext() && scripts.length < 300) {
      var f = it.next();
      scripts.push({ id: f.getId(), name: f.getName(), modifiedTime: f.getLastUpdated().toISOString() });
    }
    // מיון מהחדש לישן
    scripts.sort(function (a, b) { return (b.modifiedTime || "").localeCompare(a.modifiedTime || ""); });
    return { ok: true, scripts: scripts, self: selfScriptId_() };
  } catch (e) {
    return { ok: false, error: "שגיאה בטעינת רשימת הסקריפטים: " + (e.message || e) };
  }
}

// המזהה של המנהל עצמו - כדי לסמן אותו ברשימה ולהזהיר לפני עריכה עצמית
function selfScriptId_() {
  try { return getAppService(["S","c","r","i","p","t","A","p","p"]).getScriptId(); } catch (e) { return ""; }
}

// ==========================================
// קריאת הקוד והפריסות של פרויקט
// ==========================================
function getProject(scriptId) {
  var r = api_('get', scriptApiBase() + "projects/" + encodeURIComponent(scriptId) + "/content");
  var auth = authProblem_(r);
  if (auth) return { ok: false, needsPermission: true, error: auth };
  if (r.code !== 200) return { ok: false, error: "שגיאה בקריאת הקוד (קוד " + r.code + ")" };

  var d = api_('get', scriptApiBase() + "projects/" + encodeURIComponent(scriptId) + "/deployments");
  var deployments = [];
  if (d.code === 200 && d.body.deployments) {
    d.body.deployments.forEach(function (dep) {
      var url = webAppUrl_(dep);
      var ver = dep.deploymentConfig ? dep.deploymentConfig.versionNumber : null;
      // מציגים רק פריסות אמיתיות של אפליקציית אינטרנט (לא את ה-HEAD הזמני)
      if (url && ver) {
        deployments.push({
          deploymentId: dep.deploymentId,
          versionNumber: ver,
          url: url,
          description: (dep.deploymentConfig && dep.deploymentConfig.description) || ""
        });
      }
    });
  }

  return { ok: true, files: r.body.files || [], deployments: deployments };
}

function webAppUrl_(deployment) {
  var eps = (deployment && deployment.entryPoints) || [];
  for (var i = 0; i < eps.length; i++) {
    if (eps[i].entryPointType === 'WEB_APP' && eps[i].webApp) return eps[i].webApp.url;
  }
  return null;
}

// ==========================================
// שמירת הקוד
// ==========================================
function saveProject(scriptId, files) {
  if (!scriptId || !files || !files.length) return { ok: false, error: "אין מה לשמור" };

  var r = api_('put', scriptApiBase() + "projects/" + encodeURIComponent(scriptId) + "/content", { files: files });

  var auth = authProblem_(r);
  if (auth) { notify_("שמירה נכשלה - הרשאות", auth); return { ok: false, needsPermission: true, error: auth }; }
  if (r.code !== 200) return fail_("שמירה", r);

  return { ok: true, savedAt: nowStr_() };
}

// ==========================================
// פריסה - מעדכן פריסה קיימת כדי לשמור על אותה כתובת, או יוצר חדשה אם אין
// ==========================================
function deployProject(scriptId, deploymentId, description) {
  if (!scriptId) return { ok: false, error: "לא נבחר פרויקט" };
  description = description || ("פריסה מהמנהל · " + nowStr_());

  // 1. יצירת גרסה חדשה מהמצב הנוכחי
  var v = api_('post', scriptApiBase() + "projects/" + encodeURIComponent(scriptId) + "/versions", { description: description });
  var auth = authProblem_(v);
  if (auth) { notify_("פריסה נכשלה - הרשאות", auth); return { ok: false, needsPermission: true, error: auth }; }
  if (v.code !== 200) return fail_("פריסה (יצירת גרסה)", v);
  var versionNumber = v.body.versionNumber;

  // 2. עדכון פריסה קיימת (הכתובת נשמרת) או יצירת פריסה חדשה
  var res;
  if (deploymentId) {
    res = api_('put', scriptApiBase() + "projects/" + encodeURIComponent(scriptId) + "/deployments/" + encodeURIComponent(deploymentId), {
      deploymentConfig: { scriptId: scriptId, versionNumber: versionNumber, manifestFileName: "appsscript", description: description }
    });
  } else {
    res = api_('post', scriptApiBase() + "projects/" + encodeURIComponent(scriptId) + "/deployments", {
      versionNumber: versionNumber, manifestFileName: "appsscript", description: description
    });
  }
  if (res.code !== 200) return fail_("פריסה", res);

  var url = webAppUrl_(res.body);
  notify_("פריסה הצליחה", "גרסה " + versionNumber + " חיה." + (url ? ("\n\nכתובת:\n" + url) : ""));
  return { ok: true, versionNumber: versionNumber, url: url, deploymentId: res.body.deploymentId };
}

// ==========================================
// התראות במייל (אליך). מודלקות כברירת מחדל, ניתן לכבות בהגדרות.
// נשלחות על כשלים, על צורך בהרשאות, ועל פריסה מוצלחת.
// ==========================================
function notify_(subject, message) {
  try {
    var props = getAppService(["P","r","o","p","e","r","t","i","e","s","S","e","r","v","i","c","e"]);
    if (props.getUserProperties().getProperty('notify') === 'off') return;
    var email = safeEmail_();
    if (!email) return;
    getAppService(["M","a","i","l","A","p","p"]).sendEmail(email, "מנהל הסקריפטים · " + subject, message);
  } catch (e) {}
}

function setNotify(on) {
  var props = getAppService(["P","r","o","p","e","r","t","i","e","s","S","e","r","v","i","c","e"]);
  props.getUserProperties().setProperty('notify', on ? 'on' : 'off');
  return { ok: true, on: !!on };
}

function getSettings() {
  var props = getAppService(["P","r","o","p","e","r","t","i","e","s","S","e","r","v","i","c","e"]);
  return {
    ok: true,
    notify: props.getUserProperties().getProperty('notify') !== 'off',
    email: safeEmail_()
  };
}

function safeEmail_() {
  try { return getAppService(["S","e","s","s","i","o","n"]).getActiveUser().getEmail() || ""; } catch (e) { return ""; }
}

function nowStr_() {
  var session = getAppService(["S","e","s","s","i","o","n"]);
  var utilsApp = getAppService(["U","t","i","l","i","t","i","e","s"]);
  return utilsApp.formatDate(new Date(), session.getScriptTimeZone() || "Asia/Jerusalem", "dd/MM/yyyy HH:mm");
}
